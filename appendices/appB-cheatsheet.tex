\section{asyncio}
\label{sec:appB-asyncio}

\begin{python}[caption={The patterns}, label={lst:appB-async}]
async with asyncio.TaskGroup() as tg:          # structured; cancels siblings
    tg.create_task(work())

await asyncio.gather(*coros)                   # ordered results, orphans siblings
await asyncio.gather(*coros, return_exceptions=True)

async with asyncio.timeout(30): ...            # composes
sem = asyncio.Semaphore(5)                     # per process -- divide by replicas
q = asyncio.Queue(maxsize=100)                 # maxsize IS the backpressure
await asyncio.to_thread(blocking_io)           # blocking I/O only
async with aclosing(agen) as stream: ...       # close what you abandon

try:
    await work()
except asyncio.CancelledError:
    await cleanup()
    raise                                      # ALWAYS
\end{python}

\textbf{Middleware hook order}, measured, with \code{middleware=[A, B]}:

\begin{center}
\small
\begin{tabularx}{\linewidth}{@{}lX@{}}
\toprule
\code{before\_*} & declaration order: A, then B \\
\code{wrap\_*} & nested, first outermost: A wraps B wraps the call \\
\code{after\_*} & \textbf{reverse}: B, then A \\
\bottomrule
\end{tabularx}
\end{center}

Agent-level hooks run once per run; model-level hooks once per iteration.

\begin{warning}
Signatures that circulate wrongly: the call limit is
\code{(thread\_limit, run\_limit, exit\_behavior)} with \textbf{no}
\code{limit=}; summarisation is \code{trigger=("tokens", N)} or
\code{("messages", N)} or \code{("fraction", 0.8)} with \textbf{no}
\code{max\_tokens\_before\_summary}; PII redaction is \textbf{input-only} by
default.
\end{warning}

\section{LangGraph}
\label{sec:appB-langgraph}

\begin{python}[caption={State, graph, persistence}, label={lst:appB-lg}]
class State(TypedDict):
    messages: Annotated[list, add_messages]         # merge by id
    docs: Annotated[list[str], operator.add]        # append
    attempts: int                                    # OVERWRITE

builder = StateGraph(State, context_schema=Ctx,
                     input_schema=In, output_schema=Out)
builder.add_node("n", fn, retry_policy=RetryPolicy(max_attempts=5),
                 cache_policy=CachePolicy(ttl=60), timeout=30)  # async nodes only
builder.add_conditional_edges("n", route, ["a", "b", END])
graph = builder.compile(checkpointer=cp, store=store, cache=cache)

def router(s) -> Command[Literal["billing"]]:        # annotation is LOAD-BEARING
    return Command(update={...}, goto="billing")

def fan(s) -> list[Send]:
    return [Send("worker", {"doc": d}) for d in s["docs"]]

cfg = {"configurable": {"thread_id": "t1"}}
graph.invoke(inp, config=cfg, context=Ctx(...), durability="sync")
graph.get_state(cfg); graph.get_state_history(cfg); graph.update_state(cfg, {...})
graph.invoke(Command(resume=decision), config=cfg)

async for ns, upd in graph.astream(inp, stream_mode="updates", subgraphs=True): ...
stream = await agent.astream_events(inp, version="v3")
async for m in stream.messages:
    async for delta in m.text: ...
\end{python}

\begin{warning}
On a v3 tool call, \code{output} and \code{completed} are \textbf{mutating
attributes, not awaitables}. Iterate \code{output\_deltas} to exhaustion, then
read them.
\end{warning}

\section{.NET to Python}
\label{sec:appB-dotnet}

\begin{center}
\small
\begin{tabularx}{\linewidth}{@{}llX@{}}
\toprule
\textbf{.NET} & \textbf{Python} & \textbf{Watch out} \\
\midrule
\code{async Task<T>} & \code{async def} & The coroutine is \textbf{cold} \\
\code{Task.Run} & \code{create\_task} & Schedules; body waits for the next yield \\
\code{Task.WhenAll} & \code{gather} & Does not cancel siblings on error \\
--- & \code{TaskGroup} & Does \\
\code{CancellationToken} & \code{CancelledError} & Injected, not polled;
  re-raise \\
\code{SemaphoreSlim} & \code{Semaphore} & Per process \\
\code{IAsyncEnumerable} & \code{async def ... yield} & Close it \\
\code{await using} & \code{async with} & \\
\code{Channel<T>} & \code{Queue(maxsize=n)} & Bounded unless proven otherwise \\
\code{Task.Delay} & \code{asyncio.sleep} & \code{time.sleep} blocks the loop \\
\code{ConfigureAwait} & --- & No equivalent, none needed \\
ASP.NET middleware & agent middleware & \code{handler} is \code{next} \\
Durable Functions & LangGraph & Node = replay boundary \\
\bottomrule
\end{tabularx}
\end{center}
